     1→'use client'
     2→
     3→import { useCallback, useEffect, useRef, useState } from 'react'
     4→import { motion, AnimatePresence } from 'framer-motion'
     5→import {
     6→  ScanFace,
     7→  Camera,
     8→  ShieldCheck,
     9→  RefreshCw,
    10→  AlertTriangle,
    11→  Loader2,
    12→  Eye,
    13→  EyeOff,
    14→  User,
    15→  Lock,
    16→  CircleAlert,
    17→  UserRound,
    18→  Check,
    19→} from 'lucide-react'
    20→import { toast } from 'sonner'
    21→import type { AuthMode, FaceState } from '@/lib/aics/types'
    22→
    23→interface LoginViewProps {
    24→  onLogin: (username: string, password: string) => Promise<{ ok: boolean; error?: string }>
    25→}
    26→
    27→// ---------------------------------------------------------------------------
    28→//  Login-scoped institutional design tokens.
    29→//  Scoped to this view so the dashboard / profile keep their existing palette.
    30→//  Reference direction: real university IT portal — restrained, mature, clean.
    31→// ---------------------------------------------------------------------------
    32→const T = {
    33→  white: '#FFFFFF',
    34→  bg: '#F7F9FB', // page / help-box background
    35→  border: '#D9E0E6',
    36→  text: '#17324D', // primary text / headings
    37→  muted: '#6B7785', // secondary text / icons
    38→  primary: '#1769AA', // primary action blue
    39→  primaryDark: '#124D7A', // hover / pressed
    40→  accent: '#2F9ED8', // small accents, focus ring, scan line
    41→} as const
    42→
    43→// Whether to expose the demo / test-login shortcut (dev only).
    44→const SHOW_DEMO_LOGIN = process.env.NODE_ENV === 'development'
    45→
    46→// Dev-only demo credentials. These match the student record seeded in
    47→// MongoDB (scripts/seed-mongodb.ts) and exist ONLY so developers can
    48→// log in with one click during local development. They are NOT mock
    49→// data — the auth still hits the real MongoDB via /api/auth/login.
    50→const DEV_CREDENTIALS = { username: 'juan.santos', password: 'student123' } as const
    51→
    52→// Gradient for the "AICS Portal." accent text — top (#4EA4D7) to bottom (#64BFE9)
    53→const PORTAL_TEXT_GRADIENT = 'linear-gradient(to bottom, #4EA4D7 0%, #64BFE9 100%)'
    54→
    55→/**
    56→ * The AICS login page — a 60/40 split layout with the school image on
    57→ * the left and a white login panel (with rounded left corners) on the
    58→ * right. Supports two auth modes: Credentials (username/password) and
    59→ * Face ID (webcam-based).
    60→ *
    61→ * Visual direction: institutional / professional university IT portal.
    62→ */
    63→export function LoginView({ onLogin }: LoginViewProps) {
    64→  const [authMode, setAuthMode] = useState<AuthMode>('credentials')
    65→  const [username, setUsername] = useState('')
    66→  const [password, setPassword] = useState('')
    67→  const [showPassword, setShowPassword] = useState(false)
    68→  const [remember, setRemember] = useState(true)
    69→  const [submitting, setSubmitting] = useState(false)
    70→
    71→  const [faceState, setFaceState] = useState<FaceState>('idle')
    72→  const [faceProgress, setFaceProgress] = useState(0)
    73→  const [streamError, setStreamError] = useState<string | null>(null)
    74→  const videoRef = useRef<HTMLVideoElement | null>(null)
    75→  const streamRef = useRef<MediaStream | null>(null)
    76→  const scanTimerRef = useRef<ReturnType<typeof setInterval> | null>(null)
    77→
    78→  const stopStream = useCallback(() => {
    79→    if (streamRef.current) {
    80→      streamRef.current.getTracks().forEach((t) => t.stop())
    81→      streamRef.current = null
    82→    }
    83→    if (videoRef.current) {
    84→      videoRef.current.srcObject = null
    85→    }
    86→    if (scanTimerRef.current) {
    87→      clearInterval(scanTimerRef.current)
    88→      scanTimerRef.current = null
    89→    }
    90→  }, [])
    91→
    92→  useEffect(() => () => stopStream(), [stopStream])
    93→
    94→  const startFaceScan = useCallback(async () => {
    95→    setStreamError(null)
    96→    setFaceProgress(0)
    97→    setFaceState('starting')
    98→    try {
    99→      const stream = await navigator.mediaDevices.getUserMedia({
   100→        video: { facingMode: 'user', width: { ideal: 640 }, height: { ideal: 480 } },
   101→        audio: false,
   102→      })
   103→      streamRef.current = stream
   104→      if (videoRef.current) {
   105→        videoRef.current.srcObject = stream
   106→        await videoRef.current.play()
   107→      }
   108→      setFaceState('scanning')
   109→      let p = 0
   110→      scanTimerRef.current = setInterval(() => {
   111→        p += Math.random() * 7 + 3
   112→        if (p >= 100) {
   113→          p = 100
   114→          if (scanTimerRef.current) clearInterval(scanTimerRef.current)
   115→          setFaceProgress(100)
   116→          setFaceState('verifying')
   117→          setTimeout(() => {
   118→            setFaceState('success')
   119→            toast.success('Face verified. Welcome back to AICS Portal.')
   120→            // Face ID is a mock — log in with the dev demo credentials
   121→            setTimeout(() => {
   122→              onLogin(DEV_CREDENTIALS.username, DEV_CREDENTIALS.password)
   123→            }, 1200)
   124→          }, 1100)
   125→        } else {
   126→          setFaceProgress(p)
   127→        }
   128→      }, 220)
   129→    } catch (err) {
   130→      const e = err as DOMException
   131→      setStreamError(
   132→        e?.name === 'NotAllowedError'
   133→          ? 'Camera access was denied. Please enable camera permissions in your browser.'
   134→          : 'Unable to access camera. Please check your device and try again.'
   135→      )
   136→      setFaceState('error')
   137→      stopStream()
   138→    }
   139→  }, [stopStream, onLogin])
   140→
   141→  const cancelFaceScan = useCallback(() => {
   142→    stopStream()
   143→    setFaceState('idle')
   144→    setFaceProgress(0)
   145→    setStreamError(null)
   146→  }, [stopStream])
   147→
   148→  const retryFaceScan = useCallback(() => {
   149→    cancelFaceScan()
   150→    setTimeout(() => startFaceScan(), 50)
   151→  }, [cancelFaceScan, startFaceScan])
   152→
   153→  const handleCredentialSubmit = useCallback(
   154→    async (e: React.FormEvent) => {
   155→      e.preventDefault()
   156→      if (!username.trim() || !password.trim()) {
   157→        toast.error('Please enter both your username and password.')
   158→        return
   159→      }
   160→      setSubmitting(true)
   161→      try {
   162→        const result = await onLogin(username.trim(), password)
   163→        if (!result.ok) {
   164→          setSubmitting(false)
   165→          toast.error(result.error || 'Invalid username or password.')
   166→          return
   167→        }
   168→        toast.success('Signed in. Redirecting to your AICS dashboard...')
   169→      } catch {
   170→        setSubmitting(false)
   171→        toast.error('Network error. Please try again.')
   172→      }
   173→    },
   174→    [username, password, onLogin]
   175→  )
   176→
   177→  const handleTestLogin = useCallback(async () => {
   178→    setUsername(DEV_CREDENTIALS.username)
   179→    setPassword(DEV_CREDENTIALS.password)
   180→    setSubmitting(true)
   181→    try {
   182→      const result = await onLogin(DEV_CREDENTIALS.username, DEV_CREDENTIALS.password)
   183→      if (!result.ok) {
   184→        setSubmitting(false)
   185→        toast.error(result.error || 'Test login failed.')
   186→        return
   187→      }
   188→      toast.success('Test login successful. Welcome, Juan!')
   189→    } catch {
   190→      setSubmitting(false)
   191→      toast.error('Network error. Please try again.')
   192→    }
   193→  }, [onLogin])
   194→
   195→  return (
   196→    <main
   197→      className="w-full min-h-dvh flex flex-col lg:flex-row font-sans"
   198→      style={{ background: T.white, color: T.text }}
   199→    >
   200→      {/* ===================== LEFT 60% ===================== */}
   201→      {/*
   202→        Left panel: uses min-height (not fixed height) on large screens so it
   203→        can grow taller than the viewport when the browser is zoomed and the
   204→        right panel needs more room (flex stretches both columns to the same
   205→        height). On mobile it has a fixed hero height (42vh).
   206→        overflow-hidden only clips the decorative background image / overlay.
   207→      */}
   208→      <section
   209→        className="relative lg:w-[60%] w-full h-[42vh] lg:h-auto lg:min-h-dvh overflow-hidden flex-shrink-0"
   210→        style={{ background: T.text }}
   211→      >
   212→        {/* Campus photograph */}
   213→        <img
   214→          src="/aics-campus.jpg"
   215→          alt="Asian Institute of Computer Studies campus"
   216→          className="absolute inset-0 w-full h-full object-cover"
   217→          onError={(e) => {
   218→            ;(e.currentTarget as HTMLImageElement).style.display = 'none'
   219→          }}
   220→        />
   221→        {/* Restrained dark navy overlay — ensures text readability without saturation */}
   222→        <div
   223→          className="absolute inset-0"
   224→          style={{
   225→            background:
   226→              'linear-gradient(160deg, rgba(23,50,77,0.78) 0%, rgba(18,77,122,0.92) 100%)',
   227→          }}
   228→        />
   229→
   230→        {/* Top brand row — logo (2x) + institution name on two lines */}
   231→        <div className="absolute top-0 left-0 right-0 px-6 sm:px-10 lg:px-14 py-7 flex items-center gap-4 text-white">
   232→          <img
   233→            src="/aics-logo.svg"
   234→            alt="AICS logo"
   235→            className="flex-shrink-0 w-20 h-20 sm:w-24 sm:h-24 object-contain"
   236→          />
   237→          <div className="leading-tight">
   238→            <p className="text-base sm:text-lg font-semibold">Asian Institute</p>
   239→            <p className="text-base sm:text-lg font-semibold">of Computer Studies</p>
   240→          </div>
   241→        </div>
   242→
   243→        {/* Welcome section — vertically centered in the left panel */}
   244→        <div className="absolute inset-0 flex flex-col items-start justify-center px-6 sm:px-10 lg:px-14 text-white">
   245→          <motion.div
   246→            initial={{ opacity: 0, y: 16 }}
   247→            animate={{ opacity: 1, y: 0 }}
   248→            transition={{ duration: 0.5, ease: 'easeOut' }}
   249→          >
   250→            <p className="text-3xl sm:text-4xl lg:text-[2.75rem] font-bold leading-tight text-white">
   251→              Welcome to your
   252→            </p>
   253→            <h1
   254→              className="text-3xl sm:text-4xl lg:text-[2.75rem] font-bold leading-tight"
   255→              style={{
   256→                backgroundImage: PORTAL_TEXT_GRADIENT,
   257→                WebkitBackgroundClip: 'text',
   258→                backgroundClip: 'text',
   259→                WebkitTextFillColor: 'transparent',
   260→                color: 'transparent',
   261→              }}
   262→            >
   263→              AICS Portal.
   264→            </h1>
   265→            <p className="mt-4 text-sm sm:text-[15px] text-white max-w-md leading-relaxed">
   266→              Access your academic resources, schedules, grades, and enrollment information in one
   267→              place.
   268→            </p>
   269→          </motion.div>
   270→        </div>
   271→
   272→        {/* Footer notice — kept anchored at the bottom */}
   273→        <div className="absolute bottom-0 left-0 right-0 px-6 sm:px-10 lg:px-14 pb-10 lg:pb-14 flex items-center gap-2 text-[11px] text-white/45">
   274→          <ShieldCheck className="w-3.5 h-3.5" />
   275→          <span>Protected by AICS Information Technology Services</span>
   276→        </div>
   277→      </section>
   278→
   279→      {/* ===================== RIGHT 40% — Login panel ===================== */}
   280→      {/*
   281→        Right panel: uses min-height (100dvh) instead of fixed height (100vh)
   282→        so the panel can grow taller than the viewport when the browser is
   283→        zoomed and the form needs more room. items-center centers the form
   284→        vertically when there's space; when the form is taller than the
   285→        viewport, the section grows and the page scrolls naturally instead
   286→        of clipping. Vertical padding is always on (py-12) so content has
   287→        breathing room from the top/bottom edges even when scrolling.
   288→      */}
   289→      <section
   290→        className="relative z-10 lg:w-[calc(40%+3rem)] w-full flex-1 flex items-center justify-center px-6 sm:px-10 py-12 lg:min-h-dvh lg:-ml-12 rounded-l-[40px]"
   291→        style={{ background: T.white }}
   292→      >
   293→        <div className="relative w-full max-w-sm py-4">
   294→          {/* Heading */}
   295→          <div className="mb-8">
   296→            <h2 className="text-[22px] font-bold tracking-tight leading-snug" style={{ color: T.text }}>
   297→              Sign in to AICS Portal
   298→            </h2>
   299→            <p className="text-sm mt-2" style={{ color: T.muted }}>
   300→              Use your AICS credentials to access your student or staff portal.
   301→            </p>
   302→          </div>
   303→
   304→          {/* Auth method switcher (Credentials / Face ID) */}
   305→          <div
   306→            className="grid grid-cols-2 gap-1 p-1 rounded-[10px] mb-6"
   307→            style={{ background: T.bg, border: `1px solid ${T.border}` }}
   308→          >
   309→            <button
   310→              type="button"
   311→              onClick={() => {
   312→                setAuthMode('credentials')
   313→                if (faceState !== 'idle') cancelFaceScan()
   314→              }}
   315→              className="rounded-[8px] py-2 text-xs font-medium transition-all inline-flex items-center justify-center gap-1.5"
   316→              style={
   317→                authMode === 'credentials'
   318→                  ? { background: T.white, color: T.primary, boxShadow: '0 1px 2px rgba(23,50,77,0.08)' }
   319→                  : { color: T.muted }
   320→              }
   321→            >
   322→              <Lock className="w-3.5 h-3.5" /> Credentials
   323→            </button>
   324→            <button
   325→              type="button"
   326→              onClick={() => setAuthMode('face')}
   327→              className="rounded-[8px] py-2 text-xs font-medium transition-all inline-flex items-center justify-center gap-1.5"
   328→              style={
   329→                authMode === 'face'
   330→                  ? { background: T.white, color: T.primary, boxShadow: '0 1px 2px rgba(23,50,77,0.08)' }
   331→                  : { color: T.muted }
   332→              }
   333→            >
   334→              <ScanFace className="w-3.5 h-3.5" /> Face ID
   335→            </button>
   336→          </div>
   337→
   338→          <AnimatePresence mode="wait">
   339→            {authMode === 'credentials' ? (
   340→              <motion.form
   341→                key="cred"
   342→                initial={{ opacity: 0, x: 8 }}
   343→                animate={{ opacity: 1, x: 0 }}
   344→                exit={{ opacity: 0, x: -8 }}
   345→                transition={{ duration: 0.2 }}
   346→                onSubmit={handleCredentialSubmit}
   347→                className="space-y-4"
   348→              >
   349→                {/* Username */}
   350→                <div className="space-y-1.5">
   351→                  <label htmlFor="username" className="block text-xs font-medium" style={{ color: T.text }}>
   352→                    Username
   353→                  </label>
   354→                  <div className="relative">
   355→                    <User
   356→                      className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 pointer-events-none"
   357→                      style={{ color: T.muted }}
   358→                    />
   359→                    <input
   360→                      id="username"
   361→                      type="text"
   362→                      autoComplete="username"
   363→                      value={username}
   364→                      onChange={(e) => setUsername(e.target.value)}
   365→                      placeholder="e.g. juan.delacruz"
   366→                      className="w-full h-11 pl-10 pr-3 rounded-[8px] text-sm bg-white border outline-none transition-colors"
   367→                      style={{ borderColor: T.border, color: T.text }}
   368→                      onFocus={(e) => {
   369→                        e.currentTarget.style.borderColor = T.primary
   370→                        e.currentTarget.style.boxShadow = `0 0 0 3px ${T.accent}26`
   371→                      }}
   372→                      onBlur={(e) => {
   373→                        e.currentTarget.style.borderColor = T.border
   374→                        e.currentTarget.style.boxShadow = 'none'
   375→                      }}
   376→                    />
   377→                  </div>
   378→                </div>
   379→
   380→                {/* Password */}
   381→                <div className="space-y-1.5">
   382→                  <label htmlFor="password" className="block text-xs font-medium" style={{ color: T.text }}>
   383→                    Password
   384→                  </label>
   385→                  <div className="relative">
   386→                    <Lock
   387→                      className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 pointer-events-none"
   388→                      style={{ color: T.muted }}
   389→                    />
   390→                    <input
   391→                      id="password"
   392→                      type={showPassword ? 'text' : 'password'}
   393→                      autoComplete="current-password"
   394→                      value={password}
   395→                      onChange={(e) => setPassword(e.target.value)}
   396→                      placeholder="Enter your password"
   397→                      className="w-full h-11 pl-10 pr-10 rounded-[8px] text-sm bg-white border outline-none transition-colors"
   398→                      style={{ borderColor: T.border, color: T.text }}
   399→                      onFocus={(e) => {
   400→                        e.currentTarget.style.borderColor = T.primary
   401→                        e.currentTarget.style.boxShadow = `0 0 0 3px ${T.accent}26`
   402→                      }}
   403→                      onBlur={(e) => {
   404→                        e.currentTarget.style.borderColor = T.border
   405→                        e.currentTarget.style.boxShadow = 'none'
   406→                      }}
   407→                    />
   408→                    <button
   409→                      type="button"
   410→                      onClick={() => setShowPassword((s) => !s)}
   411→                      className="absolute right-2 top-1/2 -translate-y-1/2 p-1.5 rounded-md transition-colors hover:bg-gray-50"
   412→                      style={{ color: T.muted }}
   413→                      aria-label={showPassword ? 'Hide password' : 'Show password'}
   414→                    >
   415→                      {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   416→                    </button>
   417→                  </div>
   418→                </div>
   419→
   420→                {/* Remember me + Forgot password */}
   421→                <div className="flex items-center justify-between pt-1">
   422→                  <label className="inline-flex items-center gap-2 cursor-pointer select-none">
   423→                    <button
   424→                      type="button"
   425→                      role="checkbox"
   426→                      aria-checked={remember}
   427→                      onClick={() => setRemember((r) => !r)}
   428→                      className="w-4 h-4 rounded-[4px] flex items-center justify-center flex-shrink-0 transition-colors"
   429→                      style={
   430→                        remember
   431→                          ? { background: T.primary, border: `1.5px solid ${T.primary}` }
   432→                          : { background: T.white, border: `1.5px solid ${T.border}` }
   433→                      }
   434→                    >
   435→                      {remember && <Check className="w-3 h-3 text-white" strokeWidth={3} />}
   436→                    </button>
   437→                    <span className="text-xs" style={{ color: T.muted }}>
   438→                      Remember me
   439→                    </span>
   440→                  </label>
   441→                  <button
   442→                    type="button"
   443→                    onClick={() => toast.info('Contact the AICS IT Office to reset your password.')}
   444→                    className="text-xs font-medium transition-colors hover:underline"
   445→                    style={{ color: T.primary }}
   446→                  >
   447→                    Forgot password?
   448→                  </button>
   449→                </div>
   450→
   451→                {/* Sign In — primary action */}
   452→                <button
   453→                  type="submit"
   454→                  disabled={submitting}
   455→                  className="w-full h-11 rounded-[8px] font-semibold text-sm text-white transition-colors disabled:opacity-60 disabled:cursor-not-allowed mt-1"
   456→                  style={{ background: T.primary }}
   457→                  onMouseEnter={(e) => {
   458→                    if (!submitting) e.currentTarget.style.background = T.primaryDark
   459→                  }}
   460→                  onMouseLeave={(e) => {
   461→                    e.currentTarget.style.background = T.primary
   462→                  }}
   463→                >
   464→                  {submitting ? (
   465→                    <span className="inline-flex items-center gap-2">
   466→                      <Loader2 className="w-4 h-4 animate-spin" /> Signing in...
   467→                    </span>
   468→                  ) : (
   469→                    'Sign In'
   470→                  )}
   471→                </button>
   472→
   473→                {/* Demo / test login — development only, visually unobtrusive */}
   474→                {SHOW_DEMO_LOGIN && (
   475→                  <>
   476→                    <div className="flex items-center gap-3 pt-3">
   477→                      <div className="flex-1 h-px" style={{ background: T.border }} />
   478→                      <span className="text-[10px] uppercase tracking-wider" style={{ color: T.muted }}>
   479→                        Demo
   480→                      </span>
   481→                      <div className="flex-1 h-px" style={{ background: T.border }} />
   482→                    </div>
   483→                    <button
   484→                      type="button"
   485→                      onClick={handleTestLogin}
   486→                      disabled={submitting}
   487→                      className="w-full h-10 rounded-[8px] text-xs font-medium flex items-center justify-center gap-2 transition-colors hover:bg-gray-50"
   488→                      style={{ background: T.white, border: `1px dashed ${T.border}`, color: T.muted }}
   489→                    >
   490→                      <UserRound className="w-3.5 h-3.5" /> Test Student Login
   491→                    </button>
   492→                    <p className="text-center text-[10px]" style={{ color: T.muted }}>
   493→                      <span className="font-mono">{DEV_CREDENTIALS.username}</span>
   494→                      {' / '}
   495→                      <span className="font-mono">{DEV_CREDENTIALS.password}</span>
   496→                    </p>
   497→                  </>
   498→                )}
   499→              </motion.form>
   500→            ) : (
   501→              <motion.div
   502→                key="face"
   503→                initial={{ opacity: 0, x: 8 }}
   504→                animate={{ opacity: 1, x: 0 }}
   505→                exit={{ opacity: 0, x: -8 }}
   506→                transition={{ duration: 0.2 }}
   507→                className="space-y-4"
   508→              >
   509→                {/* Face video area */}
   510→                <div
   511→                  className="relative aspect-square w-full rounded-[12px] overflow-hidden flex items-center justify-center"
   512→                  style={{ background: T.text, border: `1px solid ${T.border}` }}
   513→                >
   514→                  {(faceState === 'starting' ||
   515→                    faceState === 'scanning' ||
   516→                    faceState === 'verifying' ||
   517→                    faceState === 'success') && (
   518→                    <video
   519→                      ref={videoRef}
   520→                      autoPlay
   521→                      playsInline
   522→                      muted
   523→                      className="absolute inset-0 w-full h-full object-cover"
   524→                      style={{ transform: 'scaleX(-1)' }}
   525→                    />
   526→                  )}
   527→
   528→                  {faceState === 'idle' && (
   529→                    <div className="text-center px-6">
   530→                      <div
   531→                        className="mx-auto w-16 h-16 rounded-full flex items-center justify-center mb-3"
   532→                        style={{ background: 'rgba(255,255,255,0.08)' }}
   533→                      >
   534→                        <ScanFace className="w-8 h-8" style={{ color: T.accent }} />
   535→                      </div>
