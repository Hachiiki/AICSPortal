     1→'use client'
     2→
     3→import { useCallback, useEffect, useRef, useState } from 'react'
     4→import { motion, AnimatePresence } from 'framer-motion'
     5→import {
     6→  ScanFace,
     7→  Camera,
     8→  ShieldCheck,
     9→  RefreshCw,
    10→  AlertTriangle,
    11→  Loader2,
    12→  Eye,
    13→  EyeOff,
    14→  User,
    15→  Lock,
    16→  CircleAlert,
    17→  UserRound,
    18→  Check,
    19→} from 'lucide-react'
    20→import { toast } from 'sonner'
    21→import type { AuthMode, FaceState } from '@/lib/aics/types'
    22→import { TEST_CREDENTIALS } from '@/lib/aics/mock-data'
    23→
    24→interface LoginViewProps {
    25→  onLogin: (username: string, password: string) => Promise<{ ok: boolean; error?: string }>
    26→}
    27→
    28→// ---------------------------------------------------------------------------
    29→//  Login-scoped institutional design tokens.
    30→//  Scoped to this view so the dashboard / profile keep their existing palette.
    31→//  Reference direction: real university IT portal — restrained, mature, clean.
    32→// ---------------------------------------------------------------------------
    33→const T = {
    34→  white: '#FFFFFF',
    35→  bg: '#F7F9FB', // page / help-box background
    36→  border: '#D9E0E6',
    37→  text: '#17324D', // primary text / headings
    38→  muted: '#6B7785', // secondary text / icons
    39→  primary: '#1769AA', // primary action blue
    40→  primaryDark: '#124D7A', // hover / pressed
    41→  accent: '#2F9ED8', // small accents, focus ring, scan line
    42→} as const
    43→
    44→// Whether to expose the demo / test-login shortcut (dev only).
    45→const SHOW_DEMO_LOGIN = process.env.NODE_ENV === 'development'
    46→
    47→// Gradient for the "AICS Portal." accent text — top (#4EA4D7) to bottom (#64BFE9)
    48→const PORTAL_TEXT_GRADIENT = 'linear-gradient(to bottom, #4EA4D7 0%, #64BFE9 100%)'
    49→
    50→/**
    51→ * The AICS login page — a 60/40 split layout with the school image on
    52→ * the left and a white login panel (with rounded left corners) on the
    53→ * right. Supports two auth modes: Credentials (username/password) and
    54→ * Face ID (webcam-based).
    55→ *
    56→ * Visual direction: institutional / professional university IT portal.
    57→ */
    58→export function LoginView({ onLogin }: LoginViewProps) {
    59→  const [authMode, setAuthMode] = useState<AuthMode>('credentials')
    60→  const [username, setUsername] = useState('')
    61→  const [password, setPassword] = useState('')
    62→  const [showPassword, setShowPassword] = useState(false)
    63→  const [remember, setRemember] = useState(true)
    64→  const [submitting, setSubmitting] = useState(false)
    65→
    66→  const [faceState, setFaceState] = useState<FaceState>('idle')
    67→  const [faceProgress, setFaceProgress] = useState(0)
    68→  const [streamError, setStreamError] = useState<string | null>(null)
    69→  const videoRef = useRef<HTMLVideoElement | null>(null)
    70→  const streamRef = useRef<MediaStream | null>(null)
    71→  const scanTimerRef = useRef<ReturnType<typeof setInterval> | null>(null)
    72→
    73→  const stopStream = useCallback(() => {
    74→    if (streamRef.current) {
    75→      streamRef.current.getTracks().forEach((t) => t.stop())
    76→      streamRef.current = null
    77→    }
    78→    if (videoRef.current) {
    79→      videoRef.current.srcObject = null
    80→    }
    81→    if (scanTimerRef.current) {
    82→      clearInterval(scanTimerRef.current)
    83→      scanTimerRef.current = null
    84→    }
    85→  }, [])
    86→
    87→  useEffect(() => () => stopStream(), [stopStream])
    88→
    89→  const startFaceScan = useCallback(async () => {
    90→    setStreamError(null)
    91→    setFaceProgress(0)
    92→    setFaceState('starting')
    93→    try {
    94→      const stream = await navigator.mediaDevices.getUserMedia({
    95→        video: { facingMode: 'user', width: { ideal: 640 }, height: { ideal: 480 } },
    96→        audio: false,
    97→      })
    98→      streamRef.current = stream
    99→      if (videoRef.current) {
   100→        videoRef.current.srcObject = stream
   101→        await videoRef.current.play()
   102→      }
   103→      setFaceState('scanning')
   104→      let p = 0
   105→      scanTimerRef.current = setInterval(() => {
   106→        p += Math.random() * 7 + 3
   107→        if (p >= 100) {
   108→          p = 100
   109→          if (scanTimerRef.current) clearInterval(scanTimerRef.current)
   110→          setFaceProgress(100)
   111→          setFaceState('verifying')
   112→          setTimeout(() => {
   113→            setFaceState('success')
   114→            toast.success('Face verified. Welcome back to AICS Portal.')
   115→            // Face ID is a mock — log in with the test student credentials
   116→            setTimeout(() => {
   117→              onLogin(TEST_CREDENTIALS.username, TEST_CREDENTIALS.password)
   118→            }, 1200)
   119→          }, 1100)
   120→        } else {
   121→          setFaceProgress(p)
   122→        }
   123→      }, 220)
   124→    } catch (err) {
   125→      const e = err as DOMException
   126→      setStreamError(
   127→        e?.name === 'NotAllowedError'
   128→          ? 'Camera access was denied. Please enable camera permissions in your browser.'
   129→          : 'Unable to access camera. Please check your device and try again.'
   130→      )
   131→      setFaceState('error')
   132→      stopStream()
   133→    }
   134→  }, [stopStream, onLogin])
   135→
   136→  const cancelFaceScan = useCallback(() => {
   137→    stopStream()
   138→    setFaceState('idle')
   139→    setFaceProgress(0)
   140→    setStreamError(null)
   141→  }, [stopStream])
   142→
   143→  const retryFaceScan = useCallback(() => {
   144→    cancelFaceScan()
   145→    setTimeout(() => startFaceScan(), 50)
   146→  }, [cancelFaceScan, startFaceScan])
   147→
   148→  const handleCredentialSubmit = useCallback(
   149→    async (e: React.FormEvent) => {
   150→      e.preventDefault()
   151→      if (!username.trim() || !password.trim()) {
   152→        toast.error('Please enter both your username and password.')
   153→        return
   154→      }
   155→      setSubmitting(true)
   156→      try {
   157→        const result = await onLogin(username.trim(), password)
   158→        if (!result.ok) {
   159→          setSubmitting(false)
   160→          toast.error(result.error || 'Invalid username or password.')
   161→          return
   162→        }
   163→        toast.success('Signed in. Redirecting to your AICS dashboard...')
   164→      } catch {
   165→        setSubmitting(false)
   166→        toast.error('Network error. Please try again.')
   167→      }
   168→    },
   169→    [username, password, onLogin]
   170→  )
   171→
   172→  const handleTestLogin = useCallback(async () => {
   173→    setUsername(TEST_CREDENTIALS.username)
   174→    setPassword(TEST_CREDENTIALS.password)
   175→    setSubmitting(true)
   176→    try {
   177→      const result = await onLogin(TEST_CREDENTIALS.username, TEST_CREDENTIALS.password)
   178→      if (!result.ok) {
   179→        setSubmitting(false)
   180→        toast.error(result.error || 'Test login failed.')
   181→        return
   182→      }
   183→      toast.success('Test login successful. Welcome, Juan!')
   184→    } catch {
   185→      setSubmitting(false)
   186→      toast.error('Network error. Please try again.')
   187→    }
   188→  }, [onLogin])
   189→
   190→  return (
   191→    <main
   192→      className="w-full min-h-dvh flex flex-col lg:flex-row font-sans"
   193→      style={{ background: T.white, color: T.text }}
   194→    >
   195→      {/* ===================== LEFT 60% ===================== */}
   196→      {/*
   197→        Left panel: uses min-height (not fixed height) on large screens so it
   198→        can grow taller than the viewport when the browser is zoomed and the
   199→        right panel needs more room (flex stretches both columns to the same
   200→        height). On mobile it has a fixed hero height (42vh).
   201→        overflow-hidden only clips the decorative background image / overlay.
   202→      */}
   203→      <section
   204→        className="relative lg:w-[60%] w-full h-[42vh] lg:h-auto lg:min-h-dvh overflow-hidden flex-shrink-0"
   205→        style={{ background: T.text }}
   206→      >
   207→        {/* Campus photograph */}
   208→        <img
   209→          src="/aics-campus.jpg"
   210→          alt="Asian Institute of Computer Studies campus"
   211→          className="absolute inset-0 w-full h-full object-cover"
   212→          onError={(e) => {
   213→            ;(e.currentTarget as HTMLImageElement).style.display = 'none'
   214→          }}
   215→        />
   216→        {/* Restrained dark navy overlay — ensures text readability without saturation */}
   217→        <div
   218→          className="absolute inset-0"
   219→          style={{
   220→            background:
   221→              'linear-gradient(160deg, rgba(23,50,77,0.78) 0%, rgba(18,77,122,0.92) 100%)',
   222→          }}
   223→        />
   224→
   225→        {/* Top brand row — logo (2x) + institution name on two lines */}
   226→        <div className="absolute top-0 left-0 right-0 px-6 sm:px-10 lg:px-14 py-7 flex items-center gap-4 text-white">
   227→          <img
   228→            src="/aics-logo.svg"
   229→            alt="AICS logo"
   230→            className="flex-shrink-0 w-20 h-20 sm:w-24 sm:h-24 object-contain"
   231→          />
   232→          <div className="leading-tight">
   233→            <p className="text-base sm:text-lg font-semibold">Asian Institute</p>
   234→            <p className="text-base sm:text-lg font-semibold">of Computer Studies</p>
   235→          </div>
   236→        </div>
   237→
   238→        {/* Welcome section — vertically centered in the left panel */}
   239→        <div className="absolute inset-0 flex flex-col items-start justify-center px-6 sm:px-10 lg:px-14 text-white">
   240→          <motion.div
   241→            initial={{ opacity: 0, y: 16 }}
   242→            animate={{ opacity: 1, y: 0 }}
   243→            transition={{ duration: 0.5, ease: 'easeOut' }}
   244→          >
   245→            <p className="text-3xl sm:text-4xl lg:text-[2.75rem] font-bold leading-tight text-white">
   246→              Welcome to your
   247→            </p>
   248→            <h1
   249→              className="text-3xl sm:text-4xl lg:text-[2.75rem] font-bold leading-tight"
   250→              style={{
   251→                backgroundImage: PORTAL_TEXT_GRADIENT,
   252→                WebkitBackgroundClip: 'text',
   253→                backgroundClip: 'text',
   254→                WebkitTextFillColor: 'transparent',
   255→                color: 'transparent',
   256→              }}
   257→            >
   258→              AICS Portal.
   259→            </h1>
   260→            <p className="mt-4 text-sm sm:text-[15px] text-white max-w-md leading-relaxed">
   261→              Access your academic resources, schedules, grades, and enrollment information in one
   262→              place.
   263→            </p>
   264→          </motion.div>
   265→        </div>
   266→
   267→        {/* Footer notice — kept anchored at the bottom */}
   268→        <div className="absolute bottom-0 left-0 right-0 px-6 sm:px-10 lg:px-14 pb-10 lg:pb-14 flex items-center gap-2 text-[11px] text-white/45">
   269→          <ShieldCheck className="w-3.5 h-3.5" />
   270→          <span>Protected by AICS Information Technology Services</span>
   271→        </div>
   272→      </section>
   273→
   274→      {/* ===================== RIGHT 40% — Login panel ===================== */}
   275→      {/*
   276→        Right panel: uses min-height (100dvh) instead of fixed height (100vh)
   277→        so the panel can grow taller than the viewport when the browser is
   278→        zoomed and the form needs more room. items-center centers the form
   279→        vertically when there's space; when the form is taller than the
   280→        viewport, the section grows and the page scrolls naturally instead
   281→        of clipping. Vertical padding is always on (py-12) so content has
   282→        breathing room from the top/bottom edges even when scrolling.
   283→      */}
   284→      <section
   285→        className="relative z-10 lg:w-[calc(40%+3rem)] w-full flex-1 flex items-center justify-center px-6 sm:px-10 py-12 lg:min-h-dvh lg:-ml-12 rounded-l-[40px]"
   286→        style={{ background: T.white }}
   287→      >
   288→        <div className="relative w-full max-w-sm py-4">
   289→          {/* Heading */}
   290→          <div className="mb-8">
   291→            <h2 className="text-[22px] font-bold tracking-tight leading-snug" style={{ color: T.text }}>
   292→              Sign in to AICS Portal
   293→            </h2>
   294→            <p className="text-sm mt-2" style={{ color: T.muted }}>
   295→              Use your AICS credentials to access your student or staff portal.
   296→            </p>
   297→          </div>
   298→
   299→          {/* Auth method switcher (Credentials / Face ID) */}
   300→          <div
   301→            className="grid grid-cols-2 gap-1 p-1 rounded-[10px] mb-6"
   302→            style={{ background: T.bg, border: `1px solid ${T.border}` }}
   303→          >
   304→            <button
   305→              type="button"
   306→              onClick={() => {
   307→                setAuthMode('credentials')
   308→                if (faceState !== 'idle') cancelFaceScan()
   309→              }}
   310→              className="rounded-[8px] py-2 text-xs font-medium transition-all inline-flex items-center justify-center gap-1.5"
   311→              style={
   312→                authMode === 'credentials'
   313→                  ? { background: T.white, color: T.primary, boxShadow: '0 1px 2px rgba(23,50,77,0.08)' }
   314→                  : { color: T.muted }
   315→              }
   316→            >
   317→              <Lock className="w-3.5 h-3.5" /> Credentials
   318→            </button>
   319→            <button
   320→              type="button"
   321→              onClick={() => setAuthMode('face')}
   322→              className="rounded-[8px] py-2 text-xs font-medium transition-all inline-flex items-center justify-center gap-1.5"
   323→              style={
   324→                authMode === 'face'
   325→                  ? { background: T.white, color: T.primary, boxShadow: '0 1px 2px rgba(23,50,77,0.08)' }
   326→                  : { color: T.muted }
   327→              }
   328→            >
   329→              <ScanFace className="w-3.5 h-3.5" /> Face ID
   330→            </button>
   331→          </div>
   332→
   333→          <AnimatePresence mode="wait">
   334→            {authMode === 'credentials' ? (
   335→              <motion.form
   336→                key="cred"
   337→                initial={{ opacity: 0, x: 8 }}
   338→                animate={{ opacity: 1, x: 0 }}
   339→                exit={{ opacity: 0, x: -8 }}
   340→                transition={{ duration: 0.2 }}
   341→                onSubmit={handleCredentialSubmit}
   342→                className="space-y-4"
   343→              >
   344→                {/* Username */}
   345→                <div className="space-y-1.5">
   346→                  <label htmlFor="username" className="block text-xs font-medium" style={{ color: T.text }}>
   347→                    Username
   348→                  </label>
   349→                  <div className="relative">
   350→                    <User
   351→                      className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 pointer-events-none"
   352→                      style={{ color: T.muted }}
   353→                    />
   354→                    <input
   355→                      id="username"
   356→                      type="text"
   357→                      autoComplete="username"
   358→                      value={username}
   359→                      onChange={(e) => setUsername(e.target.value)}
   360→                      placeholder="e.g. juan.delacruz"
   361→                      className="w-full h-11 pl-10 pr-3 rounded-[8px] text-sm bg-white border outline-none transition-colors"
   362→                      style={{ borderColor: T.border, color: T.text }}
   363→                      onFocus={(e) => {
   364→                        e.currentTarget.style.borderColor = T.primary
   365→                        e.currentTarget.style.boxShadow = `0 0 0 3px ${T.accent}26`
   366→                      }}
   367→                      onBlur={(e) => {
   368→                        e.currentTarget.style.borderColor = T.border
   369→                        e.currentTarget.style.boxShadow = 'none'
   370→                      }}
   371→                    />
   372→                  </div>
   373→                </div>
   374→
   375→                {/* Password */}
   376→                <div className="space-y-1.5">
   377→                  <label htmlFor="password" className="block text-xs font-medium" style={{ color: T.text }}>
   378→                    Password
   379→                  </label>
   380→                  <div className="relative">
   381→                    <Lock
   382→                      className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 pointer-events-none"
   383→                      style={{ color: T.muted }}
   384→                    />
   385→                    <input
   386→                      id="password"
   387→                      type={showPassword ? 'text' : 'password'}
   388→                      autoComplete="current-password"
   389→                      value={password}
   390→                      onChange={(e) => setPassword(e.target.value)}
   391→                      placeholder="Enter your password"
   392→                      className="w-full h-11 pl-10 pr-10 rounded-[8px] text-sm bg-white border outline-none transition-colors"
   393→                      style={{ borderColor: T.border, color: T.text }}
   394→                      onFocus={(e) => {
   395→                        e.currentTarget.style.borderColor = T.primary
   396→                        e.currentTarget.style.boxShadow = `0 0 0 3px ${T.accent}26`
   397→                      }}
   398→                      onBlur={(e) => {
   399→                        e.currentTarget.style.borderColor = T.border
   400→                        e.currentTarget.style.boxShadow = 'none'
   401→                      }}
   402→                    />
   403→                    <button
   404→                      type="button"
   405→                      onClick={() => setShowPassword((s) => !s)}
   406→                      className="absolute right-2 top-1/2 -translate-y-1/2 p-1.5 rounded-md transition-colors hover:bg-gray-50"
   407→                      style={{ color: T.muted }}
   408→                      aria-label={showPassword ? 'Hide password' : 'Show password'}
   409→                    >
   410→                      {showPassword ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
   411→                    </button>
   412→                  </div>
   413→                </div>
   414→
   415→                {/* Remember me + Forgot password */}
   416→                <div className="flex items-center justify-between pt-1">
   417→                  <label className="inline-flex items-center gap-2 cursor-pointer select-none">
   418→                    <button
   419→                      type="button"
   420→                      role="checkbox"
   421→                      aria-checked={remember}
   422→                      onClick={() => setRemember((r) => !r)}
   423→                      className="w-4 h-4 rounded-[4px] flex items-center justify-center flex-shrink-0 transition-colors"
   424→                      style={
   425→                        remember
   426→                          ? { background: T.primary, border: `1.5px solid ${T.primary}` }
   427→                          : { background: T.white, border: `1.5px solid ${T.border}` }
   428→                      }
   429→                    >
   430→                      {remember && <Check className="w-3 h-3 text-white" strokeWidth={3} />}
   431→                    </button>
   432→                    <span className="text-xs" style={{ color: T.muted }}>
   433→                      Remember me
   434→                    </span>
   435→                  </label>
   436→                  <button
   437→                    type="button"
   438→                    onClick={() => toast.info('Contact the AICS IT Office to reset your password.')}
   439→                    className="text-xs font-medium transition-colors hover:underline"
   440→                    style={{ color: T.primary }}
   441→                  >
   442→                    Forgot password?
   443→                  </button>
   444→                </div>
   445→
   446→                {/* Sign In — primary action */}
   447→                <button
   448→                  type="submit"
   449→                  disabled={submitting}
   450→                  className="w-full h-11 rounded-[8px] font-semibold text-sm text-white transition-colors disabled:opacity-60 disabled:cursor-not-allowed mt-1"
   451→                  style={{ background: T.primary }}
   452→                  onMouseEnter={(e) => {
   453→                    if (!submitting) e.currentTarget.style.background = T.primaryDark
   454→                  }}
   455→                  onMouseLeave={(e) => {
   456→                    e.currentTarget.style.background = T.primary
   457→                  }}
   458→                >
   459→                  {submitting ? (
   460→                    <span className="inline-flex items-center gap-2">
   461→                      <Loader2 className="w-4 h-4 animate-spin" /> Signing in...
   462→                    </span>
   463→                  ) : (
   464→                    'Sign In'
   465→                  )}
   466→                </button>
   467→
   468→                {/* Demo / test login — development only, visually unobtrusive */}
   469→                {SHOW_DEMO_LOGIN && (
   470→                  <>
   471→                    <div className="flex items-center gap-3 pt-3">
   472→                      <div className="flex-1 h-px" style={{ background: T.border }} />
   473→                      <span className="text-[10px] uppercase tracking-wider" style={{ color: T.muted }}>
   474→                        Demo
   475→                      </span>
   476→                      <div className="flex-1 h-px" style={{ background: T.border }} />
   477→                    </div>
   478→                    <button
   479→                      type="button"
   480→                      onClick={handleTestLogin}
   481→                      disabled={submitting}
   482→                      className="w-full h-10 rounded-[8px] text-xs font-medium flex items-center justify-center gap-2 transition-colors hover:bg-gray-50"
   483→                      style={{ background: T.white, border: `1px dashed ${T.border}`, color: T.muted }}
   484→                    >
   485→                      <UserRound className="w-3.5 h-3.5" /> Test Student Login
   486→                    </button>
   487→                    <p className="text-center text-[10px]" style={{ color: T.muted }}>
   488→                      <span className="font-mono">{TEST_CREDENTIALS.username}</span>
   489→                      {' / '}
   490→                      <span className="font-mono">{TEST_CREDENTIALS.password}</span>
   491→                    </p>
   492→                  </>
   493→                )}
   494→              </motion.form>
   495→            ) : (
   496→              <motion.div
   497→                key="face"
   498→                initial={{ opacity: 0, x: 8 }}
   499→                animate={{ opacity: 1, x: 0 }}
   500→                exit={{ opacity: 0, x: -8 }}
   501→                transition={{ duration: 0.2 }}
   502→                className="space-y-4"
   503→              >
   504→                {/* Face video area */}
   505→                <div
   506→                  className="relative aspect-square w-full rounded-[12px] overflow-hidden flex items-center justify-center"
   507→                  style={{ background: T.text, border: `1px solid ${T.border}` }}
   508→                >
   509→                  {(faceState === 'starting' ||
   510→                    faceState === 'scanning' ||
   511→                    faceState === 'verifying' ||
   512→                    faceState === 'success') && (
   513→                    <video
   514→                      ref={videoRef}
   515→                      autoPlay
   516→                      playsInline
   517→                      muted
   518→                      className="absolute inset-0 w-full h-full object-cover"
   519→                      style={{ transform: 'scaleX(-1)' }}
   520→                    />
   521→                  )}
   522→
   523→                  {faceState === 'idle' && (
   524→                    <div className="text-center px-6">
   525→                      <div
   526→                        className="mx-auto w-16 h-16 rounded-full flex items-center justify-center mb-3"
   527→                        style={{ background: 'rgba(255,255,255,0.08)' }}
   528→                      >
   529→                        <ScanFace className="w-8 h-8" style={{ color: T.accent }} />
   530→                      </div>
   531→                      <p className="text-white text-sm font-medium">Face Recognition</p>
   532→                      <p className="text-white/50 text-xs mt-1 max-w-[220px] mx-auto leading-relaxed">
   533→                        Click start and look directly at the camera. Your face is your password.
   534→                      </p>
   535→                    </div>
   536→                  )}
   537→
   538→                  {(faceState === 'scanning' || faceState === 'verifying') && (
   539→                    <>
   540→                      <div className="absolute inset-0 flex items-center justify-center pointer-events-none">
