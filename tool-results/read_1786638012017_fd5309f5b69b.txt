    33→// GPA computation: unit-weighted average of finalGrade
    34→function computeGPA(subjects: Subject[]): string {
    35→  let totalUnits = 0
    36→  let weightedSum = 0
    37→  for (const s of subjects) {
    38→    const grade = parseFloat(s.finalGrade)
    39→    if (!isNaN(grade)) {
    40→      totalUnits += s.units
    41→      weightedSum += grade * s.units
    42→    }
    43→  }
    44→  if (totalUnits === 0) return '-'
    45→  return (weightedSum / totalUnits).toFixed(2)
    46→}

    48→// Group subjects by term (yearLevel + academicYear + semester)
    49→interface TermGroup {
    50→  key: string
    51→  yearLevel: string
    52→  academicYear: string
    53→  semester: string
    54→  status: string
    55→  subjects: Subject[]
    56→  gpa: string
    57→  totalUnits: number
    58→}
    59→
    60→function groupByTerm(subjects: Subject[]): TermGroup[] {
    61→  const map = new Map<string, TermGroup>()
    62→  for (const s of subjects) {
    63→    const yl = s.yearLevel || 'Unknown'
    64→    const ay = s.academicYear || 'Unknown'
    65→    const sem = s.semester || 'Unknown'
    66→    const key = `${yl}|${ay}|${sem}`
    67→    if (!map.has(key)) {
    68→      map.set(key, {
    69→        key, yearLevel: yl, academicYear: ay, semester: sem,
    70→        status: s.status || 'completed', subjects: [], gpa: '-', totalUnits: 0,
    71→      })
    72→    }
    73→    map.get(key)!.subjects.push(s)
    74→  }
    75→  const groups = Array.from(map.values())
    76→  // Sort by academicYear then semester
    77→  groups.sort((a, b) => {
    78→    if (a.academicYear !== b.academicYear) return a.academicYear.localeCompare(b.academicYear)
    79→    return a.semester.localeCompare(b.semester)
    80→  })
    81→  // Compute GPA and units per term
    82→  for (const g of groups) {
    83→    const completed = g.subjects.filter((s) => s.status !== 'in-progress' && parseFloat(s.finalGrade) > 0)
    84→    g.gpa = completed.length > 0 ? computeGPA(completed) : '-'
    85→    g.totalUnits = g.subjects.reduce((sum, s) => sum + s.units, 0)
    86→  }
    87→  return groups
    88→}

   131→  })

   452→        else setError(data.error || 'Failed to load tasks')
   453→      } catch { if (!cancelled) setError('Network error') }
   454→      finally { if (!cancelled) setLoading(false) }

   457→    return () => { cancelled = true }

   461→    student.subjects.filter((s) => s.academicYear === student.academicYear && s.semester === student.semester),
   462→  [student.subjects, student.academicYear, student.semester])

   470→      else if (status === 'PENDING') pending++
   471→      else if (status === 'MISSING') missing++
   472→      else needs++
   473→    }

   483→      if (typeFilter !== 'all' && t.type !== typeFilter) return false

   485→        const { status } = computeStatus(t)

   499→        const sb = computeStatus(b).status
   500→        if (order[sa] !== order[sb]) return order[sa] - order[sb]

   527→      } else { toast.error(data.error || 'Failed to submit task.') }

   529→    finally { setSubmitting(false) }

   550→      if (el) el.scrollIntoView({ behavior: 'smooth', block: 'start' })

   552→  }, [])
   553→
   554→  if (loading) return <div className="text-slate-500 text-sm">Loading tasks…</div>
   555→  if (error) return <div className="text-red-600 text-sm">{error}</div>
   556→

   561→        <h2 className="text-lg font-semibold text-slate-900">Tasks</h2>

   564→
   565→      {/* TASK OVERVIEW CARD */}

   569→        </div>
   570→        <div className="px-6 py-5">

   701→                className="bg-white rounded-xl border border-slate-200 shadow-sm overflow-hidden">
   702→                {/* Accordion header */}
   703→                <div onClick={() => toggleCourse(subjectCode)}
   704→                  className="px-6 py-4 flex items-center justify-between gap-4 cursor-pointer hover:bg-slate-50 transition-colors">
   705→                  <div className="min-w-0">
   706→                    <div className="flex items-center gap-2">
   707→                      <span className="font-mono text-xs font-bold text-blue-700">{subjectCode}</span>
   708→                      <h3 className="text-sm font-semibold text-slate-900 truncate">{subject?.title || subjectCode}</h3>

   715→                      <span className="inline-flex items-center gap-1 px-2 py-0.5 rounded-md text-xs font-medium bg-red-50 text-red-700 border border-red-200">
   716→                        <AlertTriangle className="w-3 h-3" /> Needs attention
   717→                      </span>
   718→                    )}
   719→                    {isExpanded ? <ChevronUp className="w-4 h-4 text-slate-400" /> : <ChevronDown className="w-4 h-4 text-slate-400" />}
   720→                  </div>
   721→                </div>

   724→                  <div className="border-t border-slate-100 overflow-x-auto">
   725→                    <table className="w-full text-sm">
   726→                      <thead>
   727→                        <tr className="bg-slate-50 border-b border-slate-100">
   728→                          <th className="px-4 py-2.5 text-[11px] font-semibold uppercase tracking-wider text-slate-500 text-left">Type</th>
   729→                          <th className="px-4 py-2.5 text-[11px] font-semibold uppercase tracking-wider text-slate-500 text-left">Task</th>
   730→                          <th className="px-4 py-2.5 text-[11px] font-semibold uppercase tracking-wider text-slate-500 text-left">Due Date</th>
   731→                          <th className="px-4 py-2.5 text-[11px] font-semibold uppercase tracking-wider text-slate-500 text-left">Status / Score</th>

   733→                        </tr>
   734→                      </thead>

   740→                          const showSubmit = !task.submitted && task.score === null

   745→                              </td>

   747→                                <p className="text-sm font-medium text-slate-900">{task.title}</p>

   752→                              </td>
   753→                              <td className="px-4 py-3">
   754→                                <div className="flex flex-col gap-0.5">
   755→                                  <span className={`inline-flex items-center gap-1 px-2 py-0.5 rounded-md text-xs font-medium border w-fit ${STATUS_COLORS[status]}`}>
   756→                                    {status === 'GRADED' && <CheckCircle2 className="w-3 h-3" />}
   757→                                    {status === 'PENDING' && <Clock className="w-3 h-3" />}
   758→                                    {status === 'MISSING' && <AlertTriangle className="w-3 h-3" />}
   759→                                    {status === 'NEEDS_ATTENTION' && <Calendar className="w-3 h-3" />}
   760→                                    {sub}
   761→                                  </span>
   762→                                  <span className="text-[10px] text-slate-400">{STATUS_LABELS[status]}</span>
   763→                                </div>
   764→                              </td>
   765→                              <td className="px-4 py-3 text-right">
   766→                                {showSubmit ? (
   767→                                  <button type="button" onClick={() => setSubmitTask(task)}
   768→                                    className={`inline-flex items-center px-3 py-1.5 rounded-lg text-xs font-medium ${isMissing ? 'bg-red-50 text-red-700 border border-red-200 hover:bg-red-100' : 'bg-blue-700 text-white hover:bg-blue-800'}`}>
   769→                                    {isMissing ? 'Submit (Late)' : 'Submit'}
   770→                                  </button>
   771→                                ) : (
   772→                                  <button type="button" onClick={() => setDetailTask(task)}
   773→                                    className="p-1.5 rounded-lg text-slate-400 hover:bg-slate-100 hover:text-slate-600">
   774→                                    <MoreHorizontal className="w-4 h-4" />
   775→                                  </button>
   776→                                )}
   777→                              </td>
   778→                            </tr>
   779→                          )
   780→                        })}
   781→                      </tbody>
   782→                    </table>
   783→                  </div>
   784→                )}
   785→              </div>
   786→            )
   787→          })}
   788→        </div>
   789→      )}
   790→
   791→      {/* Submit warning modal */}
   792→      <AnimatePresence>
   793→        {submitTask && (
   794→          <SubmitWarningModal task={submitTask} onCancel={() => setSubmitTask(null)} onConfirm={handleConfirmSubmit} submitting={submitting} />

   796→      </AnimatePresence>
   797→
   798→      {/* Task details modal */}
   799→      <AnimatePresence>
   800→        {detailTask && (
   801→          <TaskDetailsModal task={detailTask} subjectName={currentSubjects.find((s) => s.code === detailTask.subjectCode)?.title || ''} onClose={() => setDetailTask(null)} />
   802→        )}
   803→      </AnimatePresence>
   804→    </motion.div>
   805→  )
   806→}

   817→function SubmitWarningModal({ task, onCancel, onConfirm, submitting }: { task: Task; onCancel: () => void; onConfirm: () => void; submitting: boolean }) {
   818→  useEffect(() => {
   819→    const handler = (e: KeyboardEvent) => { if (e.key === 'Escape') onCancel() }
   820→    document.addEventListener('keydown', handler)
   821→    return () => document.removeEventListener('keydown', handler)
   822→  }, [onCancel])
   823→
   824→  return (
   825→    <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
   826→      className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-900/50" onClick={onCancel}>
   827→      <motion.div initial={{ scale: 0.95, opacity: 0 }} animate={{ scale: 1, opacity: 1 }} exit={{ scale: 0.95, opacity: 0 }}
   828→        className="bg-white rounded-2xl shadow-2xl max-w-md w-full overflow-hidden" onClick={(e) => e.stopPropagation()}>
   829→        <div className="px-6 py-4 flex items-center justify-between border-b border-slate-100">
   830→          <div className="flex items-center gap-2">
   831→            <AlertTriangle className="w-5 h-5 text-amber-500" />
   832→            <h3 className="font-bold text-sm text-slate-900">Wait — read this first</h3>
   833→          </div>
   834→          <button type="button" onClick={onCancel} className="p-1.5 rounded-lg text-slate-400 hover:bg-slate-100"><X className="w-4 h-4" /></button>
   835→        </div>
   836→        <div className="px-6 py-5">
   837→          <p className="text-sm text-slate-600 leading-relaxed">
   838→            You should have already passed the ACTUAL paper for this activity to your adviser/professor BEFORE submitting here. Submitting in the portal without passing the physical paper will result in a penalty the next day.
   839→          </p>

   853→function TaskDetailsModal({ task, subjectName, onClose }: { task: Task; subjectName: string; onClose: () => void }) {
   854→  useEffect(() => {
   855→    const handler = (e: KeyboardEvent) => { if (e.key === 'Escape') onClose() }
   856→    document.addEventListener('keydown', handler)
   857→    return () => document.removeEventListener('keydown', handler)
   858→  }, [onClose])
   859→
   860→  const { status, sub } = computeStatus(task)
   861→
   862→  return (
   863→    <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
   864→      className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-900/50" onClick={onClose}>
   865→      <motion.div initial={{ scale: 0.95, opacity: 0 }} animate={{ scale: 1, opacity: 1 }} exit={{ scale: 0.95, opacity: 0 }}
   866→        className="bg-white rounded-2xl shadow-2xl max-w-md w-full overflow-hidden" onClick={(e) => e.stopPropagation()}>
   867→        <div className="px-6 py-4 flex items-center justify-between border-b border-slate-100">
   868→          <h3 className="font-bold text-sm text-slate-900">Task Details</h3>
   869→          <button type="button" onClick={onClose} className="p-1.5 rounded-lg text-slate-400 hover:bg-slate-100"><X className="w-4 h-4" /></button>
   870→        </div>
   871→        <div className="px-6 py-5 space-y-3">
   872→          <DetailRow label="Title" value={task.title} />
   873→          <DetailRow label="Type" value={task.type} />
   874→          <DetailRow label="Subject" value={`${task.subjectCode} — ${subjectName}`} />
   875→          <DetailRow label="Due Date" value={new Date(task.dueDate).toLocaleDateString('en-US', { year: 'numeric', month: 'long', day: 'numeric' })} />
   876→          {task.submittedAt && <DetailRow label="Submitted" value={new Date(task.submittedAt).toLocaleDateString('en-US', { year: 'numeric', month: 'long', day: 'numeric' })} />}
   877→          <DetailRow label="Status" value={STATUS_LABELS[status]} />
   878→          {task.score !== null && <DetailRow label="Score" value={`${task.score} / ${task.maxScore}`} />}
   879→          {task.feedback && <DetailRow label="Feedback" value={task.feedback} />}
   880→          {task.description && <DetailRow label="Description" value={task.description} />}
   881→        </div>
   882→        <div className="px-6 py-4 border-t border-slate-100 flex justify-end">
   883→          <button type="button" onClick={onClose} className="px-4 py-2 rounded-lg text-sm font-medium border border-slate-200 text-slate-600 hover:bg-slate-50">Close</button>
   884→        </div>
   885→      </motion.div>
   886→    </motion.div>
   887→  )
   888→}
