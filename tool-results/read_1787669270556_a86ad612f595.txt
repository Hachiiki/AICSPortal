    31→    31→// GPA computation: unit-weighted average of finalGrade (2 decimals)
    32→    32→function computeGPA(subjects: { units: number; finalGrade: string }[]): string {
    33→    33→  let totalUnits = 0
    34→    34→  let weightedSum = 0
    35→    35→  for (const s of subjects) {
    36→    36→    const grade = parseFloat(s.finalGrade)
    37→    37→    if (!isNaN(grade)) {
    38→    38→      totalUnits += s.units
    39→    39→      weightedSum += grade * s.units
    40→    40→    }
    41→    41→  }
    42→    42→  if (totalUnits === 0) return '-'
    43→    43→  return (weightedSum / totalUnits).toFixed(2)
    44→    44→}
