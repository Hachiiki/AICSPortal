    33→    33→// GPA computation: unit-weighted average of finalGrade
    34→    34→function computeGPA(subjects: Subject[]): string {
    35→    35→  let totalUnits = 0
    36→    36→  let weightedSum = 0
    37→    37→  for (const s of subjects) {
    38→    38→    const grade = parseFloat(s.finalGrade)
    39→    39→    if (!isNaN(grade)) {
    40→    40→      totalUnits += s.units
    41→    41→      weightedSum += grade * s.units
    42→    42→    }
    43→    43→  }
    44→    44→  if (totalUnits === 0) return '-'
    45→    45→  return (weightedSum / totalUnits).toFixed(2)
    46→    46→}

    48→    48→// Group subjects by term (yearLevel + academicYear + semester)
    49→    49→interface TermGroup {
    50→    50→  key: string
    51→    51→  yearLevel: string
    52→    52→  academicYear: string
    53→    53→  semester: string
    54→    54→  status: string
    55→    55→  subjects: Subject[]
    56→    56→  gpa: string
    57→    57→  totalUnits: number
    58→    58→}
    59→    59→
    60→    60→function groupByTerm(subjects: Subject[]): TermGroup[] {
    61→    61→  const map = new Map<string, TermGroup>()
    62→    62→  for (const s of subjects) {
    63→    63→    const yl = s.yearLevel || 'Unknown'
    64→    64→    const ay = s.academicYear || 'Unknown'
    65→    65→    const sem = s.semester || 'Unknown'
    66→    66→    const key = `${yl}|${ay}|${sem}`
    67→    67→    if (!map.has(key)) {
    68→    68→      map.set(key, {
    69→    69→        key, yearLevel: yl, academicYear: ay, semester: sem,
    70→    70→        status: s.status || 'completed', subjects: [], gpa: '-', totalUnits: 0,
    71→    71→      })
    72→    72→    }
    73→    73→    map.get(key)!.subjects.push(s)
    74→    74→  }
    75→    75→  const groups = Array.from(map.values())
    76→    76→  // Sort by academicYear then semester
    77→    77→  groups.sort((a, b) => {
    78→    78→    if (a.academicYear !== b.academicYear) return a.academicYear.localeCompare(b.academicYear)
    79→    79→    return a.semester.localeCompare(b.semester)
    80→    80→  })
    81→    81→  // Compute GPA and units per term
    82→    82→  for (const g of groups) {
    83→    83→    const completed = g.subjects.filter((s) => s.status !== 'in-progress' && parseFloat(s.finalGrade) > 0)
    84→    84→    g.gpa = completed.length > 0 ? computeGPA(completed) : '-'
    85→    85→    g.totalUnits = g.subjects.reduce((sum, s) => sum + s.units, 0)
    86→    86→  }
    87→    87→  return groups
    88→    88→}

    91→    91→function exportYearPDF(student: Student, term: TermGroup) {

    95→    95→
    96→    96→  // Dynamically import jspdf + autotable to avoid SSR issues
    97→    97→  Promise.all([import('jspdf'), import('jspdf-autotable')]).then(([{ jsPDF }, mod]: any) => {
    98→    98→    mod.applyPlugin(jsPDF)
    99→    99→    const doc = new jsPDF()
   100→   100→    // Header

   127→   127→    doc.setFont('helvetica', 'normal')
   128→   128→    doc.setFontSize(8)
   129→   129→    doc.text(`Generated on ${new Date().toLocaleDateString()}`, 14, finalY + 16)
   130→   130→    doc.save(filename)
   131→   131→  })
   132→   132→}
